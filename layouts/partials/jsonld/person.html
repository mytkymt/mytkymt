{{/* Person schema for the site owner. The upstream theme only emits a `WebSite` node on the homepage, so there was no machine-readable statement of *who* this site is about. Search engines and AI agents resolve a researcher to a single entity mainly through `sameAs` links to authority profiles (ORCID, Google Scholar, researchmap), so those are emitted here even when the visible icon list changes. */}} {{- $page := . -}} {{- $person := "" -}} {{- range first 1 (where (where site.Pages "Section" "authors") "Params.superuser" true) -}} {{- $person = . -}} {{- end -}} {{- with $person -}} {{- $avatar := (.Resources.ByType "image").GetMatch "*avatar*" -}} {{/* sameAs: every social link that is an external profile, plus any extra identifiers declared in params.toml under [profiles]. */}} {{- $same_as := slice -}} {{- range .Params.social -}} {{- if in (slice "http" "https") (urls.Parse .link).Scheme -}} {{- $same_as = $same_as | append .link -}} {{- end -}} {{- end -}} {{- range site.Params.profiles.same_as -}} {{- $same_as = $same_as | append . -}} {{- end -}} {{- $same_as = $same_as | uniq -}} {{- end -}}
