{{/* ScholarlyArticle schema for `publication` pages. The theme's generic `Article` partial listed only the first author, dropped the DOI even when present in front matter, and emitted an empty description. Those three fields are exactly what a citation index or an AI agent needs to link this page to the real paper, so they are all populated here. */}} {{- $page := .page -}} {{- $summary := .summary -}} {{- $featured_image := ($page.Resources.ByType "image").GetMatch "*featured*" -}} {{- $abstract := $page.Params.abstract | default $page.Params.summary | plainify -}} {{- $doi := $page.Params.doi -}} {{/* Drop bookkeeping tags so `keywords` carries topics only. */}} {{- $skip_tags := slice "Journal paper" "Conference paper" "Preprint" "Press" "Poster" "Demo" -}} {{- $topics := slice -}} {{- range $page.Params.tags -}}{{ if not (in $skip_tags .) }}{{ $topics = $topics | append . }}{{ end }}{{- end -}} {{/* Map Academic's numeric publication_types onto schema.org types. */}} {{- $type := "ScholarlyArticle" -}} {{- with $page.Params.publication_types -}} {{- $t := index . 0 -}} {{- if in (slice "3" "4" "5") $t -}}{{- $type = "ScholarlyArticle" -}}{{- end -}} {{- if eq $t "6" -}}{{- $type = "Book" -}}{{- end -}} {{- if eq $t "7" -}}{{- $type = "Chapter" -}}{{- end -}} {{- end -}}
